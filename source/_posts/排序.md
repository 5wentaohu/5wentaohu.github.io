#  归并排序
由hdu4911为例子
```C++
#include<bits/stdc++.h>
const int N=1005;
typedef long long ll;
ll a[N],b[N],cnt;
void Merge(ll l,ll mid,ll r)
{
    ll i=l;ll j=mid+1;ll t=0;
    while(i<=mid&&j<=r)
    {
        if(a[i]>a[j])
        {
            b[t++]=a[j++];
            cnt+=mid-i+1;
        }
        else b[t++]=a[i++];
    }
    while(i<=mid)b[t++]=a[i++];
    while(j<=r)b[t++]=a[j++];
    for(int i=0;i<t;i++)a[l+i]=b[i];
}
void Mergesort(ll l,ll r)
{
    if(l<r){
        ll mid=(l+r)/2;
        Mergesort(l,mid);
        Mergesort(mid+1,r);
        Merge(l,mid,r);
    }
}
int main()
{
    ll n,k;
    while(scanf("%lld%lld",&n,&k)!=EOF)
    {
        cnt=0;
        for(ll i=0;i<n;i++)
        {
            scanf("%lld",&a[i]);
        }
         Mergesort(0,n-1);
        if(cnt<=k)printf("0\n");
        else     printf("%lld",cnt-k);
    }
    return 0;
}
```
将原本大数组一步一步切分为小数组，直到数组长度为1时停止。
以升序排列的排序为例子，我们分完之后，<mark>进行区间的合并，小的放左边</mark>
<mark>大的放右边，在进行两个区间的合并的时候只需要两个零食的指针不断挑选更大值就行</mark>
不过记得最后要把剩余的数都取完，因为可能会有左边或者右边区间里的数没有取完的情况，
由于这道题求逆序对的数量，所以在合并的时候就可以留意，<mark>当i，j开始移动时，如果i指向的大于j</mark>
<mark>指向的，那么就存在了mid-i+1的逆序对</mark>，因为i的索引肯定是小于j的索引的。
最后把合并完成的数组再放回原数组。
# 快速排序
以poj2388为例子
```C++
#include<bits/stdc++.h>
using namespace std;
const int N=100010;
int a[N];
int  partion(int l,int r)
{
    int i=l-1;
    for(int j=l;j<r;j++)
    if(a[j]<a[r])
    {
        i++;
        swap(a[i],a[j]);
    }
    swap(a[i+1],a[r]);
    return i+1;
}
void quicksort(int l,int r)
{
    if(l>=r)return ;
    int pos=partion(l,r);
    quicksort(l,pos-1);
    quicksort(pos+1,r);
}
int main()
{
    int n;
    scanf("%d",&n);
    for(int i=0;i<n;i++)scanf("%d",&a[i]);
    quicksort(0,n-1);
    for(int i=0;i<n;i++)printf("%d",a[i]);
    return 0;
}
```
快速排序基本思想，是设定一个基准值，然后不断的让小于基准值的位于基准值的左边，比基准值大的位于基准值的右边，这里我们以区间最右边作为基准值，用i和j来比对，i始终指向最后一个小于基准值的元素，j遍历整个区间，遇到比基准值小的就i++，并把值交换到i所指向的位置。
最后将swap(a[i+1],a[r]);这样就把区间分为了左右两部分。
