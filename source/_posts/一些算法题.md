# 待解决的问题。
```C++
#include<bits/stdc++.h>
using namespace std;
#define rep(i, a, b) for(int i = (a); i <= (b); i ++ )

#define repp(i, a, b) for(int i = (a); i >= (b); i -- )
const int N = 2e5 + 10;
typedef long long ll;
pair<int, int> a[N];
set<int> o; multiset<ll> A, B; ll sm; int n, m, k;
void ADD(ll x)
{
    if(x == 0) return;
    if(A.size() < m + 1) { A.insert(x); sm += x; return; }
    if(*A.begin() >= x) { B.insert(x); return; }
    auto it = A.begin(); sm += x - *it; B.insert(*it); A.erase(it); A.insert(x);
}
void DEL(ll x)

{
    if(x == 0) return;
    if(B.find(x) != B.end()) { B.erase(B.find(x)); return; }
    A.erase(A.find(x)); sm -= x;
    if(B.size() > 0)
    {
        sm += *(--B.end()); A.insert(*(--B.end())); B.erase((--B.end()));
    }
}
void add(int x)
{
    auto it = o.find(x);
    int l = *prev(it), r = *next(it);
    DEL(x-l-1), DEL(r-x-1), ADD(r-l-1);
    o.erase(it);
}
void del(int x)
{
    auto it = o.upper_bound(x);
    int r = *it, l = *prev(it);
    DEL(r-l-1); ADD(r-x-1), ADD(x-l-1);
    o.insert(x);
}
void solve()
{

    cin >> n >> m >> k;
    rep(i, 1, n) cin >> a[i].first, a[i].second = i;
    sort(a + 1, a + 1 + n); o.clear(), A.clear(), B.clear();
    rep(i, 1, n) o.insert(i); o.insert(0), o.insert(n+1);
    int x = 1, y = 0, res = 1e9; sm = 0;
    while(x <= n)
    {
        while(y < n && sm < k)
        {
            int val = a[y+1].first;
            while(y < n && a[y+1].first == val) add(a[y+1].second), y ++;
        }
        if(sm < k) break;
        res = min(res, a[y].first - a[x].first);
        int val = a[x].first;
        while(x <= n && a[x].first == val) del(a[x].second), x ++;
    }
    cout << res << '\n';
}
int main()
{
    ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
    int t; cin >> t; while(t --) solve();
    return 0;
}
```
<mark>一些提示</mark>
枚举一个值域窗口 [L, R]（L,R 都是数组里出现过的值）； 把值落在 [L,R] 内的位置标出来，它们天然形成若干连续段； 若最大的 m+1 段长度之和 ≥ k，说明存在 k 个位置、分成 ≤ m+1 段、值全在 [L,R] 内 ⟹ 用 ≤ m 次翻转可行，答案候选 R-L； 求最小的 R-L。
**题目**
小 A 有一个长度为 nn 的正整数序列 aa，他定义这个序列的权值为所有长度为 kk 的区间的最小极差
小 A 可以进行至多 mm 次操作：每次操作选择一个满足 1≤l≤r≤n1≤l≤r≤n 的区间 [l,r][l,r]，将 [l,r][l,r] 这段区间翻转，也就是变为 ar,ar−1,⋯ ,alar​,ar−1​,⋯,al​。
小 A 想求出在进行不超过 mm 次翻转操作后，所有可能得到的序列中权值的最小值。
## 数位运算
```C++
#include <bits/stdc++.h>
using namespace std;
string addBinary(string &s1, string &s2) {
    reverse(s1.begin(), s1.end());
    reverse(s2.begin(), s2.end());
    int len = min(s1.size(), s2.size());
    int sum = 0;  
    for (int i = 0; i < len; i++) {
        int a = s1[i] - '0';
        int b = s2[i] - '0';
        int digit = a + b + sum;
        s1[i] = '0' + digit % 2;
        sum = digit / 2;
    }
    for (int i = len; i < (int)s1.size(); i++) {
        int a = s1[i] - '0';
        int digit = a + sum;
        s1[i] = '0' + digit % 2;
        sum = digit / 2;
    }
    while (sum > 0) {
        s1.push_back('0' + sum % 2);
        sum /= 2;
    }
    reverse(s1.begin(), s1.end());
    return s1;
}
int main() {
    string s;
    cin >> s;
    string s1 = s + "0000";  
    string s2 = s + "00";    
    string ans = addBinary(s1, s2);
    ans = addBinary(ans, s);
    cout << ans << endl;
    return 0;
}
```
在取到的字符串的末尾加上0，即为在原本数组上乘于2，在进行位运算的时候需要先将字符串翻转，因为原本输入最左边是最大位。
然后将每一位取出来，相加，用sum来处理进位，而digit处理本位，但要记得在长度较小的字符串结束之后，要继续处理长度大的字符串的进位，
最后如果sum还有则还要处理进位，然后在翻转一次再输出。
