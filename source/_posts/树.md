*关于线段树*
一种特殊的平衡二叉树，能够在数据很大的时候，节省大量的时间来进行单点修改，以及区间修改等操作
带lazy标记的线段树：
```C++
#include <vector>
using namespace std;
class SegmenTreelazy {
private:
   int n;
    vector<int> tree;  // 存储线段树节点值
    vector<int> lazy;  // 懒标记：表示当前节点需要加多少
    // 建树：从数组构建（保持不变）
    void build(const vector<int>& arr, int node, int l, int r) {
        if (l == r) {
            tree[node] = arr[l];
           return;
        }
        int mid = (l + r) / 2;
        build(arr, node*2, l, mid);
        build(arr, node*2+1, mid+1, r);
        tree[node] = tree[node*2] + tree[node*2+1]; // 求和（可改为最大/最小）
    }
    // ===== 新增：下推懒标记函数 =====
    void pushDown(int node, int l, int r) {
        if (lazy[node] != 0 && l != r) {  // 有标记且不是叶子节点
            int mid = (l + r) / 2;
            int left = node * 2;
            int right = node * 2 + 1;
            // 左孩子
            lazy[left] += lazy[node];
            tree[left] += lazy[node] * (mid - l + 1);
            // 右孩子
            lazy[right] += lazy[node];
            tree[right] += lazy[node] * (r - mid);
            // 清除当前节点标记
            lazy[node] = 0;
        }
    }
    // ===== 修改：原来是单点更新 (int idx, int delta)，现在改为区间更新 =====
    void update(int node, int l, int r, int ql, int qr, int delta) {
        // 完全覆盖：直接更新当前节点，打标记，不往下递归
        if (ql <= l && r <= qr) {
            tree[node] += delta * (r - l + 1);
            lazy[node] += delta;  // ===== 新增：打懒标记 =====
            return;
        }
        // ===== 新增：部分覆盖时先下推旧标记 =====
        pushDown(node, l, r);
        int mid = (l + r) / 2;
        if (ql <= mid) update(node*2, l, mid, ql, qr, delta);
        if (qr > mid) update(node*2+1, mid+1, r, ql, qr, delta);
        tree[node] = tree[node*2] + tree[node*2+1];
    }
    // ===== 修改：区间查询，增加了 pushDown =====
    int query(int node, int l, int r, int ql, int qr) {
        if (ql <= l && r <= qr) return tree[node];  // 完全覆盖
        // ===== 新增：部分覆盖时先下推标记 =====
        pushDown(node, l, r);
        int mid = (l + r) / 2;
        return query(node*2, l, mid, ql, qr) +
               query(node*2+1, mid+1, r, ql, qr);
    }
public:
    // ===== 修改：构造函数中新增 lazy 初始化 =====
    SegmenTreelazy(const vector<int>& arr) {
        n = arr.size();
        tree.resize(4 * n);      // 安全大小
        lazy.resize(4 * n, 0);   // ===== 新增：初始化懒标记数组 =====
        build(arr, 1, 0, n-1);
    }
    // ===== 修改：原来是单点更新 update(int idx, int delta)，现在改为区间更新 =====
    void update(int l, int r, int delta) {
        update(1, 0, n-1, l, r, delta);
    }
    // 区间查询（保持不变）
    int query(int l, int r) {
        return query(1, 0, n-1, l, r);
    }
};
```
lazy标记即是标记当前节点已经加过，但是子节点还没有加，这是一种延迟处理的方式，所以处理有lazy标记的节点时
需要将lazy标记传向子节点，并且将子节点进行更新。
